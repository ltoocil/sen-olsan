159  function stepBar(step) {


160    var items = [1, 2, 3, 4, 5].map(function (number) {


161      var cls = number < step ? "is-done" : number === step ? "is-now" : "";


162      var current = number === step ? ' aria-current="step"' : "";


163      return "<li class=\"" + cls + "\"" + current + ">" + number + "</li>";


164    }).join("");


165    return '<ol class="stepbar" aria-label="Beş bölümün ilerlemesi">' + items +


166      '<li class="step-label">Bölüm ' + step + " / 5</li></ol>";


167  }

259  function metersHtml(scenario, ids, pendingId, fromBaseline) {


260    var shownIds = pendingId ? ids.concat(pendingId) : ids.slice();


261    var shown = scoreFor(scenario, shownIds);


262    if (!shown) return "";


263    var before = fromBaseline ? shown.baseline : scoreFor(scenario, ids).current;


264    var meta = window.SEN_SCORE.meta;


265    var peopleReasons = fromBaseline ? shown.finalPeopleWhy : {};


266    var valueWhy = "";


267    if (pendingId && shown.steps.length) {


268      var last = shown.steps[shown.steps.length - 1];


269      valueWhy = last.vw || "";


270      peopleReasons = last.pw || {};


271    } else if (fromBaseline) {


272      valueWhy = shown.valueChain.filter(Boolean).join(" ");


273    }


274    var valueRows = meterRows(meta.valueLabels, before.values, shown.current.values, null);


275    var peopleRows = meterRows(meta.peopleLabels[scenario.id], before.people, shown.current.people, peopleReasons);


276    return '<div class="meters">' +


277      meterBlock("Oyun içi değer göstergeleri", "Oyun içi değer göstergeleri; bilimsel ölçüm değildir. Başlangıç sayıları bu sahne için yazılmış tabanlardır ve 5’in katlarıdır.", valueRows + (valueWhy ? '<p class="meter-why">' + esc(valueWhy) + "</p>" : "")) +


278      '<p class="meter-separate">' + esc(meta.separate) + "</p>" +


279      meterBlock("Kurmaca memnuniyet", "Kurmaca simülasyon puanı; gerçek anket sonucu değildir. Puan, o kurmaca kişinin son pratik durumdan memnuniyetidir.", peopleRows) +


280      "</div>";


281  }
